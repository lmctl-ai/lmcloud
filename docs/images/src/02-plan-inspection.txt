# Pattern: Detailed execution plan inspection and saving.
# Demonstrates running tofu.plan() in-process, inspecting planned additions/changes/destructions,
# verifying the human-readable summary string, and persisting the binary plan with plan.save().

tf = "resource \"terraform_data\" \"item1\" {\n  input = \"alpha\"\n}\nresource \"terraform_data\" \"item2\" {\n  input = \"beta\"\n}\n";
write_file(path_join(workdir, "main.tf"), tf);

p = _lmcode_ cli="lmcloud";
let dir = "build/ex02-workdir";
let ok_init = tofu.init(dir);
let plan = tofu.plan(dir);
let adds = plan.add();
let changes = plan.change();
let destroys = plan.destroy();
let summary = plan.summary();
let ok_save = plan.save("build/ex02-workdir/saved.tfplan");
end _lmcode_
